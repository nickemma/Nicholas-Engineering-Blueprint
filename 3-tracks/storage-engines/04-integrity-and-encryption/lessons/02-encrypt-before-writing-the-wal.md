# 02 Encrypt values before the WAL

> Integrity and encryption · Week 1 · Builds on lesson 01

## The problem

File permissions limit who can open an SSTable on a running host. A copied WAL or table still needs protection against exposure of the value bytes.

## The mental model

The caller passes tenant, key, and plaintext to the wrapper. The wrapper prepares a versioned header and a random nonce, then seals the value with AES-256-GCM. The engine sees only the envelope. On reads, the wrapper authenticates before returning plaintext.

## The pattern

Authenticated encryption protects value contents and detects changes. Associated data binds the envelope to its version, key ID, and physical tenant key. Load the active key and readable older keys from a file outside the database directory. Rotation changes the write key while old versions remain readable. Keys, ordering, and value lengths remain visible in storage. A previously valid envelope for the same key can still be replayed.

use std::{
    collections::BTreeMap,
    fs::File,
    io::{self, Read},
    path::Path,
};

use aes_gcm::{
    Aes256Gcm, KeyInit,
    aead::{Aead, Payload},
};

use crate::{Namespace, Row, TenantSnapshot};

const VERSION: u8 = 1;
const HEADER: usize = 5;
const NONCE: usize = 12;
const TAG: usize = 16;

pub struct Keyring {
    active: u32,
    keys: BTreeMap<u32, Aes256Gcm>,
}

impl Keyring {
    /// Read an operator-supplied file of numeric-id:hex-key lines.
    /// The caller keeps this file outside the database directory.
    pub fn load(path: &Path, active: u32) -> io::Result<Self> {
        let mut text = String::new();
        File::open(path)?.take(4097).read_to_string(&mut text)?;
        if text.len() > 4096 {
            return Err(invalid("keyring exceeds size limit"));
        }
        let mut keys = BTreeMap::new();
        for line in text.lines() {
            let (id, raw) = line
                .split_once(':')
                .ok_or_else(|| invalid("invalid keyring row"))?;
            let id: u32 = id.parse().map_err(|_| invalid("invalid key ID"))?;
            if id == 0 || raw.len() != 64 || !raw.as_bytes().iter().all(u8::is_ascii_hexdigit) {
                return Err(invalid("invalid key ID or AES-256 key length"));
            }
            let mut key = [0_u8; 32];
            for (dest, pair) in key.iter_mut().zip(raw.as_bytes().as_chunks::<2>().0) {
                *dest = u8::from_str_radix(std::str::from_utf8(pair).unwrap(), 16).unwrap();
            }
            if keys.insert(id, Aes256Gcm::new(&key.into())).is_some() {
                return Err(invalid("duplicate key ID"));
            }
        }
        if !keys.contains_key(&active) {
            return Err(invalid("active key ID unavailable"));
        }
        Ok(Self { active, keys })
    }

    fn seal(&self, physical: &[u8], value: &[u8]) -> io::Result<Vec<u8>> {
        if value.len() > 32 * 1024 - HEADER - NONCE - TAG {
            return Err(io::Error::new(
                io::ErrorKind::InvalidInput,
                "value exceeds encrypted storage limit",
            ));
        }
        let mut out = vec![VERSION];
        out.extend(self.active.to_be_bytes());
        let aad = associated_data(&out, physical);
        let mut nonce = [0; NONCE];
        getrandom::fill(&mut nonce).map_err(|_| io::Error::other("entropy unavailable"))?;
        let cipher = self
            .keys
            .get(&self.active)
            .ok_or_else(|| invalid("active key ID unavailable"))?;
        let encrypted = cipher
            .encrypt(
                &nonce.into(),
                Payload {
                    msg: value,
                    aad: &aad,
                },
            )
            .map_err(|_| invalid("value encryption failed"))?;
        out.extend(nonce);
        out.extend(encrypted);
        Ok(out)
    }

    fn open(&self, physical: &[u8], encrypted: &[u8]) -> io::Result<Vec<u8>> {
        if encrypted.len() < HEADER + NONCE + TAG || encrypted[0] != VERSION {
            return Err(invalid("unsupported or truncated value envelope"));
        }
        let id = u32::from_be_bytes(encrypted[1..HEADER].try_into().unwrap());
        let cipher = self
            .keys
            .get(&id)
            .ok_or_else(|| invalid("value key ID unavailable"))?;
        let nonce: [u8; NONCE] = encrypted[HEADER..HEADER + NONCE].try_into().unwrap();
        let aad = associated_data(&encrypted[..HEADER], physical);
        cipher
            .decrypt(
                &nonce.into(),
                Payload {
                    msg: &encrypted[HEADER + NONCE..],
                    aad: &aad,
                },
            )
            .map_err(|_| invalid("value authentication failed"))
    }
}

fn invalid(message: &str) -> io::Error {
    io::Error::new(io::ErrorKind::InvalidData, message)
}

fn associated_data(header: &[u8], physical: &[u8]) -> Vec<u8> {
    [b"switchyard/lsm/value\0".as_slice(), header, physical].concat()
}

pub struct EncryptedNamespace<'a> {
    inner: Namespace<'a>,
    keys: &'a Keyring,
}

impl<'a> EncryptedNamespace<'a> {
    pub fn new(inner: Namespace<'a>, keys: &'a Keyring) -> Self {
        Self { inner, keys }
    }

    pub fn put(&mut self, key: &[u8], value: &[u8]) -> io::Result<u64> {
        let encrypted = self.keys.seal(&self.inner.tenant.physical(key)?, value)?;
        self.inner.put(key, &encrypted)
    }

    pub fn delete(&mut self, key: &[u8]) -> io::Result<u64> {
        self.inner.delete(key)
    }

    pub fn get(&self, key: &[u8]) -> io::Result<Option<Vec<u8>>> {
        self.decrypt(key, self.inner.get(key)?)
    }

    pub fn get_at(&self, key: &[u8], pin: &TenantSnapshot) -> io::Result<Option<Vec<u8>>> {
        self.decrypt(key, self.inner.get_at(key, pin)?)
    }

    pub fn snapshot(&self) -> TenantSnapshot {
        self.inner.snapshot()
    }

    pub fn scan(
        &self,
        start: Option<&[u8]>,
        end: Option<&[u8]>,
        pin: Option<&TenantSnapshot>,
    ) -> io::Result<Vec<Row>> {
        self.inner
            .scan(start, end, pin)?
            .into_iter()
            .map(|(key, envelope)| {
                let value = self
                    .keys
                    .open(&self.inner.tenant.physical(&key)?, &envelope)?;
                Ok((key, value))
            })
            .collect()
    }

    fn decrypt(&self, key: &[u8], value: Option<Vec<u8>>) -> io::Result<Option<Vec<u8>>> {
        value
            .map(|envelope| self.keys.open(&self.inner.tenant.physical(key)?, &envelope))
            .transpose()
    }
}

                }
            }
        }
        Ok(())
    }
